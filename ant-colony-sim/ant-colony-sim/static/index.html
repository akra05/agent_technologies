multi-sim access via contexts:

# compare food collected across N sims
for ctx in all_metrics:
    snaps = ctx['snapshots']
    t = [s['tick'] for s in snaps]
    food = [s['food_collected_total'] for s in snaps]
    label = ctx.get('name', '?')
    ax.plot(t, food, label=label)
ax.legend(); ax.set_title('Food collected — all sims')
